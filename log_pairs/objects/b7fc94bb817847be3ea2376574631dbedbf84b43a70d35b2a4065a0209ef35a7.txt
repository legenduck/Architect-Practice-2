from __future__ import annotations

from datetime import date

import pytest

from parkfee.application.checkout import CheckoutService
from parkfee.domain.money import Money

from tests.conftest import FakeSeasonPassRepository, SessionFactory


@pytest.fixture
def service(pricing_config: PricingConfig) -> CheckoutService:
    return CheckoutService(pricing_config, FakeSeasonPassRepository())


def test_general_vehicle_pays_full_charge(
    service: CheckoutService, make_session: SessionFactory
) -> None:
    receipt = service.checkout(make_session(125))
    assert receipt.charge == Money(7000)
    assert receipt.discounts == ()
    assert receipt.total == Money(7000)
    assert (receipt.vat_supply, receipt.vat_tax) == (Money(6364), Money(636))


def test_reduction_then_membership_then_rounding(
    service: CheckoutService, make_session: SessionFactory
) -> None:
    """7,000원 → 경형 30% → 4,900원 → 멤버십 10% → 4,410원 → 절사 4,400원."""
    session = make_session(125, vehicle_type=VehicleType.COMPACT, is_member=True)
    receipt = service.checkout(session)
    assert receipt.discount_total == Money(2590)
    assert receipt.total == Money(4400)


def test_rounding_applies_after_discounts(
    service: CheckoutService, make_session: SessionFactory
) -> None:
    """2,500원 → 국가유공자 50% → 1,250원 → 절사 1,200원 (2026-05 정산 불일치 재발 방지)."""
    receipt = service.checkout(make_session(40, reduction=ReductionType.VETERAN))
    assert receipt.total == Money(1200)


@pytest.mark.parametrize(("lot_id", "grace"), [("HJ-01", 10), ("HJ-02", 15), ("HJ-03", 10)])
def test_exit_within_grace_period_is_free(
    service: CheckoutService, make_session: SessionFactory, lot_id: str, grace: int
) -> None:
    """요금 규정 제5조: 회차 시간 이내 출차는 무료."""
    receipt = service.checkout(make_session(grace, lot_id=lot_id))
    assert receipt.grace_applied
    assert receipt.total == Money.zero()
    assert receipt.charge == Money.zero()


@pytest.mark.parametrize(
    ("lot_id", "grace", "first_fee"), [("HJ-01", 10, 2000), ("HJ-02", 15, 1500)]
)
def test_one_minute_over_grace_is_charged(
    service: CheckoutService,
    make_session: SessionFactory,
    lot_id: str,
    grace: int,
    first_fee: int,
) -> None:
    receipt = service.checkout(make_session(grace + 1, lot_id=lot_id))
    assert not receipt.grace_applied
    assert receipt.total == Money(first_fee)

def test_coupon_policy_examples(service: CheckoutService, make_session: SessionFactory) -> None:
    """할인 규정 제13~15조: 시간 선차감과 할인 경로 비교."""
    from datetime import datetime

    from parkfee.domain.coupon import Coupon, CouponKind

    def coupon(kind: CouponKind, value: int) -> Coupon:
        return Coupon(
            "C", kind, value, "HJ-02", "매장", datetime(2026, 9, 1), datetime(2026, 9, 30)
        )

    receipt = service.checkout(make_session(660, lot_id="HJ-02"), coupon(CouponKind.TIME, 120))
    assert receipt.total == Money(27000)
    assert receipt.coupon is not None and receipt.coupon.status.value == "applied"
    receipt = service.checkout(
        make_session(120, lot_id="HJ-02", vehicle_type=VehicleType.COMPACT, is_member=True),
        coupon(CouponKind.AMOUNT, 1000),
    )
    assert receipt.total == Money(3700)
    assert receipt.coupon is not None and receipt.coupon.status.value == "unused"
    receipt = service.checkout(
        make_session(120, lot_id="HJ-02", is_member=True), coupon(CouponKind.TIME, 60)
    )
    assert receipt.total == Money(3000)
    assert receipt.coupon is not None and receipt.coupon.status.value == "applied"


def test_coupon_boundaries(
    service: CheckoutService, make_session: SessionFactory, pricing_config: PricingConfig
) -> None:
    """할인 규정 제12, 14~18조: 범위, 절사 후 동액, 기간 및 무료 출차."""
    from datetime import datetime

    from parkfee.domain.coupon import Coupon, CouponKind

    def coupon(
        value: int = 1000,
        lot: str = "HJ-01",
        start: datetime = datetime(2026, 9, 1, 10),
        end: datetime = datetime(2026, 9, 1, 11),
    ) -> Coupon:
        return Coupon("C", CouponKind.AMOUNT, value, lot, "매장", start, end)

    def status(minutes: int, item: Coupon, **kwargs: object) -> str:
        result = service.checkout(make_session(minutes, **kwargs), item).coupon
        assert result is not None
        return result.status.value

    assert status(60, coupon()) == "applied"  # 시작 포함
    assert status(120, coupon()) == "applied"  # 종료 포함
    assert status(121, coupon()) == "expired"
    assert status(59, coupon()) == "unused"
    assert status(60, coupon(lot="HJ-02")) == "unused"
    assert status(10, coupon(start=datetime(2026, 9, 1))) == "unused"
    assert status(60, coupon(value=1)) == "applied"  # 절사 후 100원 차이
    assert status(60, coupon(value=1000), reduction=ReductionType.VETERAN) == "unused"
    assert service.checkout(make_session(60), coupon(value=99999)).total == Money.zero()
    passes = FakeSeasonPassRepository(
        [SeasonPass("34나5678", "HJ-01", date(2026, 9, 1), date(2026, 9, 30))]
    )
    receipt = CheckoutService(pricing_config, passes).checkout(
        make_session(60, plate="34나5678"), coupon()
    )
    assert receipt.coupon is not None and receipt.coupon.status.value == "unused"


def test_coupon_comparison_after_rounding_and_time_remainder(
    service: CheckoutService, make_session: SessionFactory
) -> None:
    """할인 규정 제13, 15~16조: 잔여 시간과 절사 후 동액을 판단한다."""
    from datetime import datetime

    from parkfee.domain.coupon import Coupon, CouponKind

    start, end = datetime(2026, 9, 1), datetime(2026, 9, 30)
    amount = Coupon("A", CouponKind.AMOUNT, 750, "HJ-01", "매장", start, end)
    receipt = service.checkout(make_session(40, reduction=ReductionType.VETERAN), amount)
    assert receipt.total == Money(1200)
    assert receipt.coupon is not None and receipt.coupon.status.value == "unused"
    time = Coupon("T", CouponKind.TIME, 15, "HJ-02", "매장", start, end)
    receipt = service.checkout(make_session(20, lot_id="HJ-02"), time)
    assert receipt.total == Money(1500)
    assert receipt.coupon is not None and receipt.coupon.status.value == "unused"
    time = Coupon("T", CouponKind.TIME, 30, "HJ-02", "매장", start, end)
    receipt = service.checkout(make_session(20, lot_id="HJ-02"), time)
    assert receipt.total == Money.zero()
    assert receipt.coupon is not None and receipt.coupon.status.value == "applied"
